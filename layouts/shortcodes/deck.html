{{/* layouts/shortcodes/deck.html Embeds a self-contained HTML deck (or any full-viewport HTML app) inside an article. Usage in content/posts/my-article.md: {{< deck src="/decks/spike-arrest.html" full="true" >}} Params: src (required) path to the deck under static/, e.g. "/decks/spike-arrest.html" full (optional) "true" breaks out of the content column to viewport width ratio (optional) CSS aspect-ratio for the frame; default "16/9" title (optional) iframe title, for screen readers The deck file goes in static/ and is served verbatim: static/decks/spike-arrest.html → src="/decks/spike-arrest.html" */}} {{- $src := .Get "src" -}} {{- $ratio := .Get "ratio" | default "16/9" -}} {{- $title := .Get "title" | default "Interactive deck" -}} {{- $full := eq (.Get "full") "true" -}} {{- if not $src -}} {{- errorf "deck shortcode: missing required 'src' parameter (%s)" .Position -}} {{- end -}} {{/* Go's html/template CSS-escapes a bare "16/9" to ZgotmplZ (the "/" could start a comment), which silently collapses the frame to zero height. Validate the value ourselves, then mark it safe so it survives into the style attribute. */}} {{- if not (findRE `^[0-9]+(\.[0-9]+)?\s*/\s*[0-9]+(\.[0-9]+)?$` $ratio) -}} {{- errorf "deck shortcode: ratio %q must look like \"16/9\" (%s)" $ratio .Position -}} {{- end -}} {{- $ratioCSS := printf "aspect-ratio:%s" $ratio | safeCSS -}} {{/* per-page counter: gives each deck a unique id and lets us emit CSS/JS only once, regardless of how many other shortcodes appear on the page */}} {{- $n := .Page.Scratch.Get "deckCount" | default 0 -}} {{- .Page.Scratch.Set "deckCount" (add $n 1) -}} {{- $id := printf "deck-%d" $n -}} {{- if eq $n 0 }} {{- end }}
